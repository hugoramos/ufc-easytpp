\section{Introdução}

\begin{frame}{Sequências de eventos do mundo real}
  Muitos fenômenos acontecem em tempo contínuo e precisamos modelá-los:
  \vspace{0.4em}
  \begin{columns}[T]
    \begin{column}{0.25\textwidth}
      \centering
      \textbf{Telemetria} \\
      \small falhas de sistema, alertas de infraestrutura
    \end{column}
    \begin{column}{0.25\textwidth}
      \centering
      \textbf{Redes sociais} \\
      \small retweets, curtidas, cascatas de respostas
    \end{column}
    \begin{column}{0.25\textwidth}
      \centering
      \textbf{Saúde} \\
      \small internações, exames, prescrições de um paciente
    \end{column}
    \begin{column}{0.25\textwidth}
      \centering
      \textbf{Manutenção} \\
      \small falhas em equipamentos industriais
    \end{column}
  \end{columns}
  \vspace{0.8em}
  \begin{block}{O que eles têm em comum?}
    Cada domínio produz uma sequência ordenada de eventos com instantes de tempo:
    \[
      \mathcal{S} = \{(t_1, k_1), (t_2, k_2), \ldots, (t_N, k_N)\}
    \]
    onde $t_i$ é o tempo e $k_i$ é o tipo (marca) do evento $i$.
  \end{block}
\end{frame}

\begin{frame}{O que queremos modelar?}
  Dados os eventos passados, queremos responder:
  \begin{itemize}
    \item \textbf{Quando} o próximo evento vai acontecer?
    \item \textbf{De qual tipo} ele será?
  \end{itemize}
  \vspace{0.5em}
  A ferramenta central é a \textbf{intensidade condicional}:
  \[
    \lambda(t \mid \mathcal{H}_t) = \lim_{\Delta t \to 0}
      \frac{P(\text{evento em } [t, t+\Delta t) \mid \mathcal{H}_t)}{\Delta t}
  \]
  onde $\mathcal{H}_t$ é o histórico de todos os eventos antes de $t$.
  \vspace{0.5em}
  \begin{block}{Objetivo de aprendizado}
    Aprender $\lambda(t)$ a partir dos dados de modo que o modelo generalize para
    sequências com comprimentos diferentes dos vistos no treino.
  \end{block}
\end{frame}

\begin{frame}{O problema prático: treinar curto, testar longo}
  \begin{itemize}
    \item Coletar sequências longas de eventos costuma ser caro ou inviável.
    \item Queremos \textbf{treinar em sequências curtas} e rodar o modelo em
          sequências \textbf{arbitrariamente mais longas} \cite{press2022train}.
    \item Modelos de processos pontuais baseados em Transformer degradam bastante
          quando a sequência de teste é mais longa que a de treino. A capacidade
          de se manter estável nesse regime é chamada de
          \emph{extrapolação de comprimento}.
  \end{itemize}
  \vspace{0.6em}
  \begin{block}{Contribuição deste trabalho: HoTHP}
    Um modelo da família Transformer que embute um \textbf{viés de recência
    exponencial} (eventos recentes pesam mais), espelhando a estrutura do
    processo de Hawkes, e por isso mantém a verossimilhança estável quando a
    sequência de teste é muito mais longa que a de treino.
    {\small (\emph{Como} esse viés entra na atenção, vemos na metodologia.)}
  \end{block}
\end{frame}

\begin{frame}{Objetivo e pergunta de pesquisa}
  \begin{block}{Pergunta de pesquisa}
    No mecanismo de atenção (detalhado adiante), substituir um kernel que
    \textbf{oscila} por um que \textbf{decai exponencialmente} melhora a capacidade de extrapolação de comprimento dos
    processos pontuais temporais neurais?
  \end{block}
  \vspace{0.5em}
  Para responder, nós:
  \begin{itemize}
    \item propomos e implementamos o \textbf{HoTHP}, adaptando o HoPE de posições
          discretas para instantes de tempo contínuos;
    \item avaliamos sob \emph{treinar curto, testar longo} em dados
          \textbf{sintéticos} (processo gerador conhecido) e dados \textbf{reais}
          de domínios variados;
    \item analisamos \emph{sob quais condições} o viés de recência é benéfico,
          neutro ou prejudicial (intensidades aprendidas, acurácia, RMSE e uma
          ablação de escala temporal).
  \end{itemize}
\end{frame}

\section{Fundamentação: Processos Pontuais Temporais}

\begin{frame}{A distribuição de Poisson}
  \textbf{A pergunta que ela responde:} quantos eventos veremos em uma janela
  fixa, se eles acontecem de forma independente a uma taxa média constante?
  \vspace{0.3em}
  \begin{block}{Definição \cite{ross_first_2019}}
    Se o \emph{número esperado} de eventos na janela é $\lambda > 0$, a
    probabilidade de observar exatamente $k$ eventos é
    \[
      \Pr(X = k) = \frac{\lambda^{k} e^{-\lambda}}{k!}, \qquad k = 0, 1, 2, \ldots
    \]
    \vspace{-0.6em}
    {\footnotesize
    \begin{itemize}
      \item $X$: uma \textbf{variável aleatória}, a contagem de eventos na janela
            (desconhecida antes de observarmos);
      \item $k$: um \textbf{valor concreto} que essa contagem pode assumir ($0, 1, 2, \ldots$);
      \item $\Pr(X = k)$: a \textbf{probabilidade} de a contagem observada $X$
            ser exatamente $k$.
    \end{itemize}}
  \end{block}
  {\small
  \begin{itemize}
    \item $\lambda$ é ao mesmo tempo a \textbf{média} e a \textbf{variância} da contagem.
  \end{itemize}}
\end{frame}

\begin{frame}{O processo de Poisson}
  Agora deixe os eventos acontecerem ao longo do tempo, a uma \textbf{taxa}
  constante $\lambda$ (eventos por unidade de tempo), e seja $N(T)$ a contagem
  deles até o tempo $T$ \cite{ross_first_2019}. De forma resumida:
  \begin{itemize}
    \item eventos em intervalos curtos e \textbf{sem sobreposição} acontecem de
          forma \textbf{independente};
    \item cada intervalo curto recebe \textbf{no máximo um} evento, com chance
          proporcional a $\lambda$.
  \end{itemize}
  Então $N(T)$ é Poisson com média $\lambda T$ (taxa vezes tempo):
  \[
    \Pr\{N(T) = n\} = \frac{(\lambda T)^{n}}{n!}\, e^{-\lambda T}
  \]
  \vspace{-0.3em}
  \begin{center}
    \includegraphics[width=0.56\textwidth]{figura-poisson}
  \end{center}
\end{frame}

\begin{frame}{Processos Pontuais Temporais (TPP)}
  Modelos probabilísticos para sequências de eventos discretos em
  \textbf{tempo contínuo} \cite{zuo2020transformer}. Eles lidam com eventos em
  tempos irregulares e arbitrários.
  \vspace{0.3em}
  \begin{itemize}
    \item Totalmente descritos pela intensidade condicional $\lambda(t \mid \mathcal{H}_t)$.
    \item Com \textbf{marcas} (tipos de evento), a sequência é
          $X = \{(t_1, m_1), \ldots, (t_N, m_N)\}$ e usamos $K$ intensidades:
  \end{itemize}
  \[
    \lambda_k(t \mid \mathcal{H}_t) = \lim_{\Delta t \to 0}
    \frac{\Pr(\text{evento do tipo } k \text{ em } [t, t+\Delta t) \mid \mathcal{H}_t)}{\Delta t}
  \]
  \begin{block}{Sobre a palavra ``marca''}
    Usamos \emph{marca} e \emph{tipo} como sinônimos: a marca é um tipo de evento
    categórico, uma dentre $K$ classes.
  \end{block}
\end{frame}

\begin{frame}{Função de perda: a log-verossimilhança}
  Dados todos os eventos observados em $[0,T]$, os TPPs são treinados maximizando
  a log-verossimilhança \cite{mei2017neural}:
  \[
    \mathcal{L} = \sum_{i:\, t_i \leq T} \log \lambda_{k_i}(t_i)
      \;-\; \underbrace{\int_{0}^{T} \bar{\lambda}(t)\, dt}_{\Lambda},
    \qquad \bar{\lambda}(t) = \sum_{k=1}^{K} \lambda_k(t)
  \]
  \begin{itemize}
    \item Primeiro termo: recompensa uma \textbf{intensidade alta} nos momentos em
          que os eventos realmente ocorreram (e para o tipo correto $k_i$).
    \item Segundo termo (o \emph{compensador} $\Lambda$): penaliza massa de
          intensidade onde nenhum evento ocorreu.
    \item O treino minimiza a log-verossimilhança \textbf{negativa} (NLL), nossa
          principal métrica de avaliação.
  \end{itemize}
  Na prática, a integral é estimada numericamente, exatamente da mesma forma para
  todos os modelos que comparamos.
\end{frame}

\begin{frame}{A intuição da verossimilhança: um orçamento de intensidade}
  Os dois termos competem: subir a intensidade nos eventos reais \textbf{recompensa}
  (1º termo, $\sum \log \lambda$), mas \textbf{toda} a área sob a curva é
  \textbf{penalizada} (2º termo, o compensador $\Lambda$).
  \vspace{0.2em}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \centering
      \textbf{\small Intensidade alta e chapada}\\[0.2em]
      \begin{tikzpicture}[scale=0.66,>=stealth]

      {\scriptsize \textbf{Área grande}: boa recompensa, mas paga muita penalidade.}
    \end{column}

  \vspace{0.2em}
  \begin{block}{O orçamento de intensidade}
    Os dois recebem recompensa parecida nos eventos, mas o da direita gasta muito
    menos intensidade \emph{fora} deles. A verossimilhança premia \textbf{concentrar}
    a intensidade nos instantes certos, não espalhá-la.
  \end{block}
\end{frame}

\begin{frame}{O processo de Hawkes}
  Poisson supõe eventos independentes. O \textbf{processo de Hawkes} supõe que
  \textbf{eventos passados aumentam a probabilidade de eventos futuros}, com uma
  influência que decai exponencialmente \cite{hawkes1971spectra}:
  \[
    \lambda(t \mid \mathcal{H}_t) = \mu + \sum_{t_i < t} \alpha \, e^{-\beta (t - t_i)}
  \]
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \begin{itemize}
        \item $\mu$: intensidade de base.
        \item $\alpha$: salto instantâneo causado por um evento.
        \item $\beta$: quão rápido essa influência decai.
      \end{itemize}
      \vspace{0.3em}
      O decaimento exponencial é um \textbf{viés de recência}: eventos recentes
      importam mais.
    \end{column}
    \begin{column}{0.46\textwidth}
      \centering
      \includegraphics[width=\textwidth]{figura-hawkes}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{O processo de Cox, e onde o Hawkes se encaixa}
  Formas cada vez mais ricas de especificar a intensidade:
  \vspace{0.4em}
  \begin{enumerate}
    \item \textbf{Poisson homogêneo:} a intensidade $\lambda$ é constante.
    \item \textbf{Poisson não homogêneo:} $\lambda(t)$ varia no tempo, mas de
          forma determinística.
    \item \textbf{Processo de Cox} (Poisson duplamente estocástico): $\lambda(t)$
          é ele próprio um processo \emph{aleatório}, guiado por fatores externos
          aos eventos.
    \item \textbf{Processo de Hawkes:} $\lambda(t)$ é aleatório porque depende dos
          \emph{eventos passados do próprio processo}, cada um elevando
          temporariamente a intensidade.
  \end{enumerate}
  \vspace{0.5em}
  \begin{block}{Por que isso importa aqui}
    O Hawkes também tem intensidade estocástica, mas, diferente do Cox, sua
    aleatoriedade é \textbf{endógena}: $\lambda(t)$ é um funcional do próprio
    passado do processo (um processo \emph{auto-excitante}, não duplamente
    estocástico). É exatamente essa dependência do histórico, com decaimento
    exponencial, que motiva o viés de recência do nosso modelo.
  \end{block}
\end{frame}

\section{Modelos neurais e atenção}

\begin{frame}{Neural Hawkes Process (NHP)}
  O processo de Hawkes clássico é limitado: a influência é sempre excitatória,
  constante por tipo, e decai exponencialmente. Mei e Eisner \cite{mei2017neural}
  relaxam essas suposições com redes neurais.
  \begin{itemize}
    \item Permitem $\alpha, \mu \in \mathbb{R}$, de modo que os eventos podem
          \textbf{inibir} além de excitar; um \textbf{softplus} mantém a
          intensidade positiva.
    \item Substituem a soma aditiva por uma \textbf{LSTM de tempo contínuo}
          (CT-LSTM): a célula de memória decai exponencialmente entre eventos.
  \end{itemize}
  \[
    \lambda_k(t) = f_k\!\left(\mathbf{w}_k^\top \mathbf{h}(t)\right), \qquad
    \mathbf{c}(t) = \bar{\mathbf{c}} + (\mathbf{c} - \bar{\mathbf{c}})\,
      e^{-\bm{\delta}(t - t_i)}
  \]
  Um modelo simples, mas forte: ainda é um dos melhores benchmarks hoje, e uma
  baseline importante nos nossos experimentos.
\end{frame}

\begin{frame}{Transformers e o mecanismo de atenção}
  RNNs (incluindo a CT-LSTM) processam eventos \textbf{sequencialmente}, o que é
  lento e tem dificuldade com sequências longas. O Transformer
  \cite{vaswani2017attention} elimina a recorrência: todos os eventos são
  processados de uma vez pela \textbf{atenção}.
  \[
    \text{Attention}(Q,K,V) = \text{softmax}\!\left(\frac{QK^\top}{\sqrt{d_k}}\right) V
  \]
  \begin{itemize}
    \item Cada evento presta atenção a todos os outros; os pesos medem a relevância.
    \item A atenção \textbf{multi-cabeça} roda $h$ projeções em paralelo e as
          concatena.
  \end{itemize}
  \vspace{-0.2em}
  \begin{figure}
    \centering
    \includegraphics[height=0.29\textheight,keepaspectratio]{figura-attention}

    {\tiny\color{gray} Fonte: \cite{vaswani2017attention}}
  \end{figure}
\end{frame}

\begin{frame}{Codificação posicional: a atenção não tem ordem}
  A atenção é \textbf{equivariante a permutações}: reordenar as entradas apenas
  reordena as saídas, então, sozinha, ela não conhece a ordem nem o tempo dos
  eventos. A posição precisa ser injetada.
  \vspace{-0.2em}
  \begin{columns}[T]
    \begin{column}{0.54\textwidth}
      \small
      \begin{itemize}
        \item \textbf{Transformer original:} soma uma codificação senoidal da
              \emph{posição discreta} a cada embedding de token.
        \item Cada dimensão é um seno/cosseno de uma frequência diferente: rápida
              à esquerda, lenta à direita, com assinatura única por posição.
        \item \textbf{Para TPPs} precisamos do \emph{instante} de fato, não só da
              ordem, já que o momento do evento importa.
      \end{itemize}
    \end{column}
    \begin{column}{0.44\textwidth}
      \centering
      \includegraphics[width=\textwidth,height=0.58\textheight,keepaspectratio]{fig_positional_encoding}

      {\scriptsize Codificação senoidal: posição (linhas) vs. dimensão (colunas).}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Transformer Hawkes Process (THP)}
  Zuo et al. \cite{zuo2020transformer} adaptam o Transformer para TPPs com uma
  \textbf{codificação temporal} que usa o instante $t_j$ em vez de uma posição
  discreta:
  \[
    [z(t_j)]_i =
    \begin{cases}
      \cos\!\left(t_j / 10000^{(i-1)/M}\right), & i \text{ ímpar},\\[2pt]
      \sin\!\left(t_j / 10000^{i/M}\right), & i \text{ par}.
    \end{cases}
  \]
  \vspace{-0.8em}
  \begin{columns}[T]
    \begin{column}{0.54\textwidth}
      A codificação é somada ao embedding de tipo, e então passa por autoatenção
      e uma rede feed-forward. A intensidade é
      \[
        \resizebox{\linewidth}{!}{$
        \lambda_k(t) = f_k\!\Big(
          \underbrace{\alpha_k \tfrac{t-t_j}{t_j}}_{\text{atual}}
          + \underbrace{\mathbf{w}_k^\top \mathbf{h}(t_j)}_{\text{histórico}}
          + \underbrace{b_k}_{\text{base}}\Big)
        $.}
      \]
    \end{column}
    \begin{column}{0.44\textwidth}
      \centering
      \includegraphics[width=0.80\textwidth,height=0.47\textheight,keepaspectratio]{figura-thp}\\[0.2em]
      {\tiny\color{gray} Fonte: \cite{zuo2020transformer}}
    \end{column}
  \end{columns}
\end{frame}

\section{Trabalhos relacionados: extrapolação de comprimento}

\begin{frame}{Extrapolação de comprimento em modelos de linguagem}
  Um modelo treinado em sequências de comprimento $L_{\text{train}}$ deveria se
  manter estável quando testado em $L_{\text{test}} > L_{\text{train}}$
  \cite{press2022train}.
  \vspace{0.3em}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \begin{itemize}
        \item Posições senoidais absolutas produzem padrões de codificação nunca vistos no treino para índices maiores: a perplexidade explode.
        \item Codificações rotativas (RoPE) ajudam ao usar posições
              \emph{relativas}, mas Press et al. \cite{press2022train} mostraram
              que elas ainda falham para sequências muito mais longas.
        \item Só o \textbf{ALiBi}, com uma penalidade linear de distância, se
              mantém estável bem além do comprimento de treino.
      \end{itemize}
    \end{column}
    \begin{column}{0.46\textwidth}
      \centering
      \includegraphics[width=\textwidth,height=0.56\textheight,keepaspectratio]{fig_alibi_extrapolation}\\[0.2em]
      {\tiny\color{gray} Fonte: \cite{press2022train}}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{O mesmo problema em TPPs: treinar curto, testar longo}
  Adaptamos o protocolo \emph{treinar curto, testar longo} \cite{press2022train}
  para processos pontuais, usando o \textbf{número de eventos} como comprimento.
  \vspace{0.4em}
  \begin{itemize}
    \item Treinar em sequências de comprimento fixo (por exemplo, $L = 50$ eventos).
    \item Testar em sequências $\alpha$ vezes mais longas, $\alpha \in \{1, 2, 5, 10\}$.
    \item Em $\alpha = 10\times$: treinar com 50 eventos, testar com 500.
  \end{itemize}
  \vspace{0.5em}
  \begin{block}{O que medimos}
    Se a NLL se mantém estável conforme $\alpha$ cresce. Um modelo com um viés de
    recência adequado deve degradar muito pouco.
  \end{block}
\end{frame}

\begin{frame}{Rotary Position Embedding (RoPE)}
  Su et al. \cite{su2024roformer} notaram que a atenção só precisa da posição
  \textbf{relativa}. O RoPE rotaciona pares de dimensões do embedding por um
  ângulo proporcional à posição:
  \[
    \mathbf{R}_j(\Delta) =
    \begin{bmatrix} \cos(\theta_j \Delta) & \sin(\theta_j \Delta) \\
                   -\sin(\theta_j \Delta) & \cos(\theta_j \Delta) \end{bmatrix},
    \qquad \theta_j = 10000^{-2(j-1)/d}.
  \]
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{itemize}
        \item Aplicado a $\mathbf{q}$ e $\mathbf{k}$ antes do produto interno.
        \item O score acaba dependendo apenas da diferença de posições.
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \centering
      \includegraphics[width=0.85\textwidth,height=0.53\textheight,keepaspectratio]{fig_rope_to_hope}\\[0.2em]
      {\tiny\color{gray} Adaptado de \cite{su2024roformer}}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{RoTHP: RoPE para processos pontuais}
  Dai et al. \cite{dai2026rothp} trazem o RoPE para TPPs: rotacionar pelo
  \textbf{instante} em vez do índice discreto.
  \vspace{0.4em}
  \begin{itemize}
    \item O score de atenção depende apenas de $\Delta t = t_i - t_j$.
    \item Isso torna a verossimilhança \textbf{invariante no tempo}: deslocar
          todos os tempos por uma constante $\sigma$ não altera o resultado.
    \item Robusto a ruído nos instantes (dessincronização de relógio), comum em
          dados reais. Novo estado da arte em vários benchmarks de TPP.
  \end{itemize}
  \vspace{0.4em}
  \begin{block}{Mas há um porém}
    O kernel rotativo é construído a partir de $\cos$ e $\sin$: ele
    \textbf{oscila}. Essa é a limitação que nosso método ataca.
  \end{block}
\end{frame}

\begin{frame}{ALiBi e o viés indutivo de recência}
  Press et al. \cite{press2022train} seguem um caminho diferente: nenhum
  embedding, apenas uma \textbf{penalidade linear} fixa no score de atenção:
  \[
    \text{softmax}\!\left(\mathbf{q}_i \mathbf{K}^\top
      + m \cdot [-(i{-}1), \ldots, -2, -1, 0]\right)
  \]
  \begin{itemize}
    \item A penalidade cresce com a distância, então tokens distantes são
          suprimidos. A inclinação $m$ é fixa por cabeça, nunca aprendida.
    \item Isso é um \textbf{viés indutivo em direção à recência}, embutido na
          arquitetura em vez de aprendido dos dados.
  \end{itemize}
  \begin{block}{A conexão com o Hawkes}
    É exatamente a ideia do kernel de Hawkes (a influência
    $\propto e^{-\beta \Delta t}$ decai com a distância). O THP e o RoTHP abriram
    mão dessa garantia estrutural ao trocar o kernel por atenção livre.
    \textbf{O HoTHP a devolve.}
  \end{block}
\end{frame}

\section{Metodologia: HoTHP}

\begin{frame}{A limitação oscilatória do RoTHP}
  O score do RoTHP depende apenas de $\Delta t = t_i - t_j$, mas seu kernel é
  construído a partir de $\cos$ e $\sin$, então ele \textbf{oscila} em vez de
  decair.
  \begin{columns}[T]
    \begin{column}{0.44\textwidth}
      \begin{itemize}
        \item O \textbf{RoPE} (azul) decai apenas \emph{em média}: ruidoso, não
              monótono, e nunca some de fato.
        \item Um kernel de Hawkes $\varphi(\Delta t)=\alpha e^{-\beta\Delta t}$ é
              \emph{estritamente} decrescente. O RoPE viola isso.
        \item Na extrapolação a fase pode cair em um \textbf{pico}: um evento
              distante recebe score maior que um próximo.
      \end{itemize}
      \vspace{0.2em}

    \end{column}
    \begin{column}{0.56\textwidth}
      \centering
      \includegraphics[width=\textwidth,height=0.72\textheight,keepaspectratio]{figura-oscilacao}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Do kernel trigonométrico ao kernel hiperbólico}
  Seguindo o HoPE \cite{dai2025hope}, substituímos o bloco de rotação
  $\mathbf{R}_j$ por um \textbf{boost hiperbólico} $\mathbf{B}_j$:
  \[
    \resizebox{0.78\textwidth}{!}{$
    \mathbf{R}_j(\Delta t) =
      \begin{bmatrix} \cos & \sin \\ -\sin & \cos \end{bmatrix}
    \;\longrightarrow\;
    \mathbf{B}_j(\Delta t) =
      \begin{bmatrix} \cosh(\theta_j \Delta t) & \sinh(\theta_j \Delta t) \\
                      \sinh(\theta_j \Delta t) & \cosh(\theta_j \Delta t) \end{bmatrix}
    $}
  \]
  \vspace{-0.4em}
  \begin{center}
    \includegraphics[width=0.92\textwidth,height=0.46\textheight,keepaspectratio]{fig_rope_vs_hothp_geom}
  \end{center}
  \vspace{-0.4em}
  {\footnotesize A rotação preserva o \textbf{círculo} ($\cos,\sin$ oscilam); o
  boost de Lorentz preserva a \textbf{hipérbole} ($\cosh,\sinh$ são monótonos).
  Controlamos esse crescimento com o fator de decaimento
  $e^{-|\Delta t|\theta'}$.}
\end{frame}

\begin{frame}{Decaimento exponencial e o parâmetro $\theta'$}
  O HoPE \cite{dai2025hope} adiciona um $\theta'$ aprendível e amortece
  $\mathbf{q}, \mathbf{k}$ para que o score decaia. Em tempo contínuo:
  \[
    g(\Delta t) = e^{-|\Delta t|\,\theta'}\; \mathbf{q}^\top \mathbf{B}(\theta, \Delta t)\, \mathbf{k}
  \]
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{itemize}
        \item Se $\theta' > \max_j \theta_j$, o decaimento domina o crescimento
              hiperbólico: o score \textbf{decai exponencialmente até zero} com
              $|\Delta t|$, sem oscilação.
        \item Esse é um viés de recência \textbf{estrutural}: vale mesmo para
              $\Delta t$ nunca visto no treino. É isso que permite a extrapolação.
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \centering
      \includegraphics[width=\textwidth]{fig_decay_hiperbolico}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Score do HoTHP: um par de dimensões por vez}
  Cada cabeça se divide em $d/2$ \textbf{pares}. O par $p$ tem frequência
  $\theta_p$ e componentes $(q_1^{(p)}, q_2^{(p)})$ e $(k_1^{(p)}, k_2^{(p)})$.
  \vspace{0.3em}
  \begin{block}{Score de um par $=$ rotação hiperbólica por $\Delta t$}
    \[
      s^{(p)} = (q_1 k_1 + q_2 k_2)\,\cosh(\theta_p \Delta t)
              + (q_1 k_2 + q_2 k_1)\,\sinh(\theta_p \Delta t)
    \]
  \end{block}
  Score completo $=$ soma sobre os pares, com o fator de recência
  $e^{-|\Delta t|\theta'}$ e o fator usual $1/\sqrt{d}$ ($\Delta t = t_i - t_j$):
  \[
    s_{ij} = \frac{1}{\sqrt{d}} \sum_{p=1}^{d/2}
      e^{-|\Delta t|\theta'}\; s^{(p)} .
  \]

\end{frame}

\begin{frame}{Nunca avaliamos $\cosh/\sinh$, apenas exponenciais}
  \textbf{Passo 1.} Substituímos cada um pela sua definição ($\cosh$ é par, então
  $\theta_p\Delta t \to \theta_p|\Delta t|$):
  \[
    \cosh(\theta_p \Delta t) = \tfrac{1}{2}\big(e^{\theta_p|\Delta t|} + e^{-\theta_p|\Delta t|}\big),
    \qquad
    \sinh(\theta_p \Delta t) = \tfrac{\mathrm{sign}(\Delta t)}{2}\big(e^{\theta_p|\Delta t|} - e^{-\theta_p|\Delta t|}\big).
  \]
  \textbf{Passo 2.} Multiplicamos o termo do $\cosh$ pelo fator e \emph{juntamos
  os expoentes} (esse é todo o truque):
  \begin{align*}
    e^{-|\Delta t|\theta'}\cosh(\theta_p \Delta t)
      &= \tfrac{1}{2}\Big(
          e^{-|\Delta t|\theta'}\, e^{+\theta_p|\Delta t|}
        + e^{-|\Delta t|\theta'}\, e^{-\theta_p|\Delta t|}\Big) \\[0.3em]
      &= \tfrac{1}{2}\Big(
          e^{-|\Delta t|(\theta' - \theta_p)}
        + e^{-|\Delta t|(\theta' + \theta_p)}\Big).
  \end{align*}
  \textbf{Passo 3.} Como $\theta' > \max_p \theta_p$ por construção, tanto
  $\theta'-\theta_p>0$ quanto $\theta'+\theta_p>0$: \alert{todo expoente é
  negativo}, então cada termo fica em sem overflow, sem \texttt{NaN}.
  \vspace{0.2em}
  {\small (O termo do $\sinh$ se junta da mesma forma, resultando em
  $\tfrac{\mathrm{sign}(\Delta t)}{2}\big(e^{-|\Delta t|(\theta'-\theta_p)} - e^{-|\Delta t|(\theta'+\theta_p)}\big)$.)}
\end{frame}

\begin{frame}{Arquitetura: onde o viés exponencial atua}
  O HoTHP é um encoder Transformer com atenção multi-cabeça hiperbólica.
  \textbf{Nenhuma} informação de tempo é somada ao embedding; o tempo entra
  apenas pelo kernel. O viés atua em \textbf{duas etapas distintas}:
  \vspace{0.3em}
  \begin{enumerate}
    \item \textbf{Atenção hiperbólica} $\to \mathbf{h}(t_j)$: o decaimento decide
          \emph{quais eventos passados} moldam o estado oculto. Eventos recentes
          recebem mais peso.
    \item \textbf{Intensidade} $\lambda_k(t) = \mathrm{softplus}\!\big(
          \alpha_k (t-t_j) + \mathbf{w}_k^\top \mathbf{h}(t_j) + b_k\big)$:
          linear no tempo \emph{decorrido} $t-t_j$, então deslocar todos os
          instantes não muda nada; $\alpha_k$ é livre (pode subir ou descer após
          um evento).
  \end{enumerate}
  \begin{block}{Consequência}
    O viés de recência \textbf{não} exige que o processo seja auto-excitante.
    Exige apenas que eventos recentes sejam mais informativos. Ele ajuda em
    processos auto-excitantes \emph{e} auto-inibitórios.
  \end{block}
\end{frame}

\begin{frame}{Kernel RoTHP vs. HoTHP: zonas de treino e extrapolação}
  \begin{figure}
    \centering
    \includegraphics[width=0.74\textwidth]{fig_kernel_rothp_vs_hothp}
  \end{figure}
  \vspace{-0.3em}
  {\small Na \textbf{zona de treino}, os dois modelos ajustam os dados. Na

\end{frame}

\section{Experimentos e resultados}

  \begin{itemize}
    \item \textbf{Dados sintéticos:} Hawkes multivariado com kernel exponencial,
          dois regimes: \emph{decaimento lento} ($\beta$ pequeno) e
          \emph{decaimento rápido} ($\beta$ grande). Conhecemos o processo gerador.
    \item \textbf{Dados reais (EasyTPP \cite{xue2024easytpp}):} Taxi, StackOverflow,

          mesma arquitetura e intensidade, diferindo \textbf{apenas em como o
          tempo entra na atenção} (uma comparação controlada).
    \item \textbf{Métricas:} NLL (principal), RMSE, acurácia de tipo. 
  \end{itemize}
\end{frame}

  \begin{itemize}
    \item \textbf{Notação e legibilidade:} siglas expandidas no resumo, lista de
          símbolos adicionada, notação unificada ($\gamma_k$ para a inclinação da
          intensidade, $\rho$ para o fator de extrapolação) e o módulo corrigido
          na equação do kernel ($e^{-|\Delta t|\theta'}$).
    \item \textbf{Fundamentação mais profunda:} uma taxonomia Poisson--Hawkes, a
          ligação explícita de verossimilhança Poisson\,$\to$\,Hawkes e uma
          discussão sobre a limitação de período do RoPE.
    \item \textbf{Mais resultados:} RMSE agora reportado nos conjuntos reais, e
          fontes adicionadas a toda figura reaproveitada.
    \item \textbf{Escrita:} a conclusão e o capítulo de trabalhos futuros
          concluídos.
  \end{itemize}
\end{frame}

\begin{frame}{Contribuições}
  \begin{itemize}
    \item \textbf{HoTHP:} um Transformer Hawkes Process com kernel rotativo
          \textbf{hiperbólico}, combinando invariância no tempo com um viés de
          recência exponencial estrutural em um único mecanismo de atenção.
    \item Um $\theta'$ aprendível, parametrizado de modo que
          $\theta' > \max_j \theta_j$ \textbf{por construção}, garantindo o
          decaimento exponencial do score e estabilidade numérica.
    \item Sob \emph{treinar curto, testar longo}, o HoTHP mantém a NLL estável até
          $10\times$ nos dados sintéticos (significativo por Wilcoxon), e é o
          melhor da família Transformer na Amazon e no Taxi.
    \item Uma análise separando \emph{onde} o viés atua (atenção vs. intensidade),
          explicando por que ele ajuda mesmo em processos auto-inibitórios.
    \item Implementação e experimentos no framework EasyTPP.
  \end{itemize}
\end{frame}

\begin{frame}{Limitações e trabalhos futuros}
  \textbf{Limitações}
  \begin{itemize}
    \item O viés exponencial atrapalha quando eventos distantes são tão relevantes
          quanto os recentes (memória de longo alcance).
    \item Escalas temporais extremas precisam de normalização para evitar
          saturação numérica (como no Retweet cru).
    \item $\theta'$ é uma taxa de decaimento \textbf{global única}; processos com
          múltiplas escalas temporais podem exigir múltiplas taxas.
  \end{itemize}
  \vspace{-0.2em}
  \textbf{Trabalhos futuros}
  \begin{itemize}
    \item \textbf{Múltiplas escalas de decaimento:} um $\theta'$ aprendível por
          cabeça (ao estilo do ALiBi), cada uma se especializando em uma escala
          temporal.
    \item \textbf{Formas alternativas de decaimento:} lei de potência em vez de
          exponencial, para memória de longo alcance (réplicas sísmicas, cascatas).
    \item \textbf{Além da atenção:} o viés de recência se transfere para modelos
          de espaço de estados (Mamba) e TPPs baseados em LLM?
  \end{itemize}
\end{frame}
